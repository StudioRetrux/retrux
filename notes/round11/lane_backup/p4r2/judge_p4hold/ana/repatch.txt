+/**
+ * P4 R8-2 (lane p4hold; `?panokeep=1` / `?p4hold=0`: AA8 keeps the decoded panorama instead): the environment panorama again,
+ * for the PMREM after a WebGL context restore, decoded exactly as loadHeroAssets' `exr("panorama", ...)` does. `force-cache` takes
+ * the browser's cached copy when it has one (no revalidation), so the restore normally needs no network; the engine no longer
+ * keeps 4 MiB of half floats in memory for the page's whole life for a rare event.
+ */
+export async function reloadPanorama(base: string) {
+  const res = await fetch(base + ASSETS.panorama, { cache: "force-cache" });
+  if (!res.ok) throw new Error(`[hero] ${ASSETS.panorama}: HTTP ${res.status}`);
+  const loader = new EXRLoader().setDataType(THREE.HalfFloatType);
+  const d = loader.parse(await res.arrayBuffer()) as { width: number; height: number; data: Uint16Array; format: THREE.PixelFormat; type: THREE.TextureDataType };
+  const t = new THREE.DataTexture(d.data, d.width, d.height, d.format, d.type);
+  t.colorSpace = THREE.LinearSRGBColorSpace;
+  t.mapping = THREE.EquirectangularReflectionMapping;
+  t.minFilter = THREE.LinearFilter;
+  t.generateMipmaps = false;
+  t.magFilter = THREE.LinearFilter;
+  t.flipY = false;
+  t.needsUpdate = true;
